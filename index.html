  <div class="container" id="slots-container"></div>


  <script>

    const MAX_CAPACITY = 5;

    const SLOTS = ["14:00", "15:00", "16:00", "17:00", "18:00", "19:00"];


    function getBookings() {

      const stored = localStorage.getItem('cervinia_bookings');

      if (stored) return JSON.parse(stored);

      

      const initial = {};

      SLOTS.forEach(slot => initial[slot] = []);

      return initial;

    }


    function saveBookings(bookings) {

      localStorage.setItem('cervinia_bookings', JSON.stringify(bookings));

      render();

    }


    function addBooking(slot, inputId) {

      const input = document.getElementById(inputId);

      const name = input.value.trim();

      if (!name) return;


      const bookings = getBookings();

      if (bookings[slot].length < MAX_CAPACITY) {

        bookings[slot].push(name);

        saveBookings(bookings);

        input.value = '';

      }

    }


    function removeBooking(slot, index) {

      const bookings = getBookings();

      bookings[slot].splice(index, 1);

      saveBookings(bookings);

    }


    function render() {

      const bookings = getBookings();

      const container = document.getElementById('slots-container');

      container.innerHTML = '';


      SLOTS.forEach((slot, index) => {

        const list = bookings[slot] || [];

        const isFull = list.length >= MAX_CAPACITY;

        const available = MAX_CAPACITY - list.length;

        const inputId = `input-${index}`;


        const card = document.createElement('div');

        card.className = 'slot-card';

        card.innerHTML = `

          <div class="slot-header">

            <span class="time">Ore ${slot}</span>

            <span class="badge ${isFull ? 'full' : ''}">

              ${isFull ? 'Completo' : `${available} posti liberi`}

            </span>

          </div>


          <div class="form-group">

            <input type="text" id="${inputId}" placeholder="Scrivi il tuo nome" ${isFull ? 'disabled' : ''}>

            <button onclick="addBooking('${slot}', '${inputId}')" ${isFull ? 'disabled' : ''}>Prenota</button>

          </div>


          <ul class="attendees-list">

            ${list.map((name, i) => `

              <li class="attendee-item">

                <span>👤 ${name}</span>

                <button class="btn-cancel" onclick="removeBooking('${slot}', ${i})">Cancella</button>

              </li>

            `).join('')}

          </ul>

        `;

        container.appendChild(card);

      });

    }


    render();

  </script>
