function isDateValid(year, month, day) { const date = new Date(year, month - 1, day); // Prima data in assoluto consentita: 31 Ottobre 2026 const firstAvailableDate = new Date(2026, 9, 31); firstAvailableDate.setHours(0, 0, 0, 0); if (date < firstAvailableDate) return false; // Formattazione data "AAAA-MM-GG" per i controlli const dateStr = `${year}-${String(month).padStart(2, '0')}-${String(day).padStart(2, '0')}`; // 1. DATE BLOCCATE (Sintassi corretta con le virgole) const blockedDates = [ "2026-12-25", "2026-12-31", "2027-01-01" ]; if (blockedDates.includes(dateStr)) { return false; // Giorno bloccato tassativamente } // 2. DATE EXTRA AGGIUNTIVE const extraDates = [ "2026-12-08", // Martedì Immacolata ]; if (extraDates.includes(dateStr)) { return true; // Rende il giorno prenotabile anche se infrasettimanale } // 3. Festività natalizie (es. dal 23 Dic al 8 Gen) const isXmas = (month === 12 && day >= 23) || (month === 1 && day <= 8); if (isXmas) return true; // 4. Fine settimana (Sabato e Domenica) const dayOfWeek = date.getDay(); return dayOfWeek === 0 || dayOfWeek === 6; }
